import os
from langchain_community.document_loaders import PyPDFLoader
from langchain_openai import ChatOpenAI
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import StrOutputParser

# ==========================================
# 1. CONFIGURATION
# ==========================================

# Replace these with your specific details
os.environ["OPENAI_API_KEY"] = "sk-..."  # Your API Key
CUSTOM_BASE_URL = "https://your-abstracted-api.com/v1" # Your Abstracted API URL

# File Paths
PLATFORM_DOCS_PATH = "platform_docs.pdf" # Your documentation with diagrams/info
AUDIT_TRAIL_PATH = "audit_trail.pdf"     # The specific request log you want to analyze

# Initialize the LLM
# We use a large context model (like gpt-4-turbo or 32k context models) 
# because we are stuffing all docs in.
llm = ChatOpenAI(
    base_url=CUSTOM_BASE_URL,
    model="gpt-4", # Or your specific model name
    temperature=0
)

# ==========================================
# 2. DATA LOADING (No Vector Store)
# ==========================================

def load_pdf_text(file_path):
    """
    Loads a PDF and combines all pages into a single string.
    Using PyPDFLoader for reliability.
    """
    loader = PyPDFLoader(file_path)
    pages = loader.load()
    
    # Merge all pages into one large text block, keeping page numbers for reference
    full_text = ""
    for page in pages:
        full_text += f"\n--- Page {page.metadata.get('page_label', page.metadata.get('page', 'Unknown'))} ---\n"
        full_text += page.page_content
    
    return full_text

print(f"Loading Platform Docs from {PLATFORM_DOCS_PATH}...")
platform_context = load_pdf_text(PLATFORM_DOCS_PATH)

print(f"Loading Audit Trail from {AUDIT_TRAIL_PATH}...")
audit_trail_content = load_pdf_text(AUDIT_TRAIL_PATH)

# ==========================================
# 3. PHASE 1: THE INSIGHT GENERATOR
# ==========================================

print("\nRunning Phase 1: Analyzing Request State...")

insight_prompt = ChatPromptTemplate.from_template("""
You are a Senior Support Engineer analyzing a stuck request.

I have provided the PLATFORM DOCUMENTATION below. Use this to interpret the AUDIT TRAIL.

<PLATFORM_DOCUMENTATION>
{platform_context}
</PLATFORM_DOCUMENTATION>

<AUDIT_TRAIL_TO_ANALYZE>
{audit_trail}
</AUDIT_TRAIL_TO_ANALYZE>

YOUR TASK:
1. Analyze the Audit Trail step-by-step.
2. Compare the steps against the Platform Documentation to understand the flow.
3. Determine the current state of the request (e.g., Stuck, Failed, Waiting).
4. Identify the likely root cause (e.g., Timeout, Missing Data, Logic Gap).

Output your analysis clearly.
""")

# Create the chain
analysis_chain = insight_prompt | llm | StrOutputParser()

# Invoke the chain
analysis_result = analysis_chain.invoke({
    "platform_context": platform_context,
    "audit_trail": audit_trail_content
})

print("--- ANALYSIS RESULT ---")
print(analysis_result)


# ==========================================
# 4. PHASE 2: THE META-ANALYSIS (Feedback Loop)
# ==========================================

print("\nRunning Phase 2: Identifying Useful Chunks & Gaps...")

meta_prompt = ChatPromptTemplate.from_template("""
You just performed an analysis on an audit trail. Now I need you to critique your own logic and the documentation quality.

<PROVIDED_PLATFORM_DOCS>
{platform_context}
</PROVIDED_PLATFORM_DOCS>

<YOUR_ANALYSIS>
{analysis_result}
</YOUR_ANALYSIS>

Based on the above, answer these two questions strictly:

1. **USEFUL KNOWLEDGE**: Quote the specific text or sections from the <PROVIDED_PLATFORM_DOCS> that were most critical in forming your analysis. (Mention Page numbers if available).

2. **MISSING INFORMATION**: What specific details were present in the Audit Trail but NOT explained in the Platform Docs? (e.g., "The audit trail mentions 'Error Code 503' but the docs do not list error codes", or "The audit trail shows a 'VerifierService' but the docs don't explain what that service does").
""")

# Create the meta chain
meta_chain = meta_prompt | llm | StrOutputParser()

# Invoke the meta chain
meta_result = meta_chain.invoke({
    "platform_context": platform_context,
    "analysis_result": analysis_result
})

print("--- META ANALYSIS ---")
print(meta_result)
